# Dig Mini-Game Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the one-tap dig with a hot/cold search mini-game played on a 5×5 grid at the spot, reached by watching the grub crawl there.

**Architecture:** The rules stay pure (`BurrowRules`-style static functions over `(save, rng)` returning `BurrowResult`), and the grid lives in the save so progress survives. New dig code goes in its own files (`DigTypes`, `DigGen`, `DigRules`, `DigGridView`, `DigTripView`) rather than growing the four existing Burrow files. Spot positions become layout-derived so the rules' soil layers can never disagree with the drawn bands.

**Tech Stack:** Swift 5.9, SwiftUI, XCTest, XcodeGen. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-11-10-dig-minigame-design.md` — read it before starting. The parent spec is `docs/superpowers/specs/2026-11-10-burrow-mechanics-design.md`.

```swift
import Foundation

/// One tile of a dig grid. `rock` needs `hp` strikes before it breaks.
enum DigTile: Codable, Equatable {
    case soil
    case rock(hp: Int)
    case root
    case broken

    /// Can a strike land here at all? (Roots and already-broken tiles can't.)
    var isDiggable: Bool {
        switch self {
        case .soil, .rock: return true
        case .root, .broken: return false
        }
    }

    /// Strikes still needed to break this tile.
    var cost: Int {
        switch self {
        case .soil: return 1
        case .rock(let hp): return hp
        case .root, .broken: return 0
        }
    }
}

/// A per-spot twist, rolled when the find is buried. `nil` means a plain spot.
enum DigCondition: String, Codable, CaseIterable, Equatable {
    case wet, rocky, rooty, glinting
}

/// How deep the spot is. Fixed per spot; never saved (it comes from `BurrowSpots`).
enum SoilLayer: String, CaseIterable, Equatable {
    case loose, clay, deep

    /// Strikes a trip to this layer grants, before conditions and mood.
    var strikes: Int {
        switch self {
        case .loose: return 6
        case .clay:  return 5
        case .deep:  return 4
        }
    }

    /// Obstacles the layer places, beyond any the condition adds.
    var rocks: Int { self == .clay ? 3 : 0 }
    var roots: Int { self == .deep ? 3 : 0 }
}

/// What a broken tile tells you about where the find is.
enum Heat: String, Codable, Equatable { case hot, warm, cold }

/// Geometry of the 5 × 5 dig grid. Tiles are indexed row-major, row 0 at the top.
enum DigGrid {
    static let side = 5
    static let tileCount = side * side

    static func row(_ i: Int) -> Int { i / side }
    static func col(_ i: Int) -> Int { i % side }

    /// Manhattan distance between two tile indices.
    static func distance(_ a: Int, _ b: Int) -> Int {
        abs(row(a) - row(b)) + abs(col(a) - col(b))
    }

    /// Spec §1: d == 1 is hot, d == 2 is warm (cold when wet), further is cold.
    static func heat(distance d: Int, wet: Bool) -> Heat {
        switch d {
        case 1: return .hot
        case 2: return wet ? .cold : .warm
        default: return .cold
        }
    }
}
```

```swift
extension BurrowLayout {
    /// Row boundaries of the four soil bands drawn by `BurrowArt.backdrop`.
    /// soil1 = (surface+1 ..< b1), soil2 = (b1 ..< b2), soil3 = (b2 ..< b3), soil4 = (b3 ..< rows).
    var soilBands: (b1: Int, b2: Int, b3: Int) {
        let soil = rows - surface - 1
        let b1 = surface + 1 + Int(Double(soil) * 0.24)
        let b2 = b1 + Int(Double(soil) * 0.27)
        let b3 = b2 + Int(Double(soil) * 0.24)
        return (b1, b2, b3)
    }

    /// Spec §1: loose = soil1, clay = soil2, deep = soil3. soil4 holds no spots.
    func rows(for layer: SoilLayer) -> Range<Int> {
        let b = soilBands
        switch layer {
        case .loose: return (surface + 1)..<b.b1
        case .clay:  return b.b1..<b.b2
        case .deep:  return b.b2..<b.b3
        }
    }
}
```

```swift
import Foundation

/// The fixed places where finds get buried (spec §1: 8 spots in open soil).
/// Index = `BuriedFind.spot` = `BurrowSave.holes` entry. **Order is part of the save
/// format: append new spots, never reorder.**
///
/// A spot is defined by its design column, its soil layer, and how far down that layer's
/// band it sits (`t`, 0 = top of the band, 1 = bottom). Resolving against the layout rather
/// than a fixed row offset is what keeps a spot inside its layer on every screen — fixed
/// offsets overflowed the much shorter iPad burrow.
enum BurrowSpots {
    struct Spot {
        let x: Int           // design column, 0..<64
        let layer: SoilLayer
        let t: Double        // 0...1 down the layer's band
    }

    static let spots: [Spot] = [
        Spot(x: 6,  layer: .clay,  t: 0.30),   // 0  left of the chamber
        Spot(x: 39, layer: .loose, t: 0.55),   // 1  upper soil, middle
        Spot(x: 59, layer: .loose, t: 0.80),   // 2  upper soil, right edge
        Spot(x: 34, layer: .deep,  t: 0.35),   // 3  below the chamber
        Spot(x: 20, layer: .loose, t: 0.25),   // 4  upper soil, above the crawl tunnel
        Spot(x: 5,  layer: .clay,  t: 0.75),   // 5  far left, above the left side-room
        Spot(x: 62, layer: .clay,  t: 0.50),   // 6  far right, beside the right connector
        Spot(x: 42, layer: .deep,  t: 0.70),   // 7  deep soil, left of the right side-room
    ]

    static func layer(_ spot: Int) -> SoilLayer { spots[spot].layer }

    /// The spot's cell in bitmap coordinates, for the given layout.
    static func resolve(_ spot: Int, _ L: BurrowLayout) -> (x: Int, y: Int) {
        let s = spots[spot]
        let band = L.rows(for: s.layer)
        let span = max(1, band.count - 1)
        let y = band.lowerBound + Int((Double(span) * s.t).rounded())
        return (L.x0 + s.x, min(y, band.upperBound - 1))
    }
}
```

Draw order is fixed by the spec (find tile → layer obstacles → condition extras → glinting), because the save migration replays it from a seed.

```swift
import Foundation

/// Builds dig grids. The only place a grid is created — `bury`, `BurrowSave.fresh` and the
/// v1→v2 migration all come through here, so a grid is generated exactly once and then saved.
enum DigGen {
    struct Result: Equatable {
        var tiles: [DigTile]
        var findTile: Int
        /// True when the clue constraint forced obstacles to be dropped (see `make`).
        var usedFallback: Bool
    }

    static let maxAttempts = 40

    /// Spec §1 "Grid generation". RNG draw order is part of the contract.
    static func make(layer: SoilLayer, condition: DigCondition?,
                     rng: inout some RandomNumberGenerator) -> Result {
        let findTile = Int.random(in: 0..<DigGrid.tileCount, using: &rng)
        let rocks = layer.rocks + (condition == .rocky ? 3 : 0)
        let roots = layer.roots + (condition == .rooty ? 3 : 0)

        for _ in 0..<maxAttempts {
            var tiles = place(rocks: rocks, roots: roots, findTile: findTile, rng: &rng)
            applyGlint(&tiles, findTile: findTile, condition: condition, rng: &rng)
            if hasEnoughClues(tiles, findTile: findTile, wet: condition == .wet) {
                return Result(tiles: tiles, findTile: findTile, usedFallback: false)
            }
        }

        // Deterministic fallback: drop obstacles one at a time until the grid is readable.
        // It always terminates — with no obstacles, even a corner find has 5 tiles at d ≤ 2
        // and 2 at d == 1.
        var r = rocks, t = roots
        while true {
            if t > 0 { t -= 1 } else if r > 0 { r -= 1 }
            var tiles = place(rocks: r, roots: t, findTile: findTile, rng: &rng)
            applyGlint(&tiles, findTile: findTile, condition: condition, rng: &rng)
            if hasEnoughClues(tiles, findTile: findTile, wet: condition == .wet) || (r == 0 && t == 0) {
                return Result(tiles: tiles, findTile: findTile, usedFallback: true)
            }
        }
    }

    /// Roots never cover the find; rocks may (that is rocky's cost).
    private static func place(rocks: Int, roots: Int, findTile: Int,
                              rng: inout some RandomNumberGenerator) -> [DigTile] {
        var tiles = [DigTile](repeating: .soil, count: DigGrid.tileCount)
        var free = Array(0..<DigGrid.tileCount).filter { $0 != findTile }
        free.shuffle(using: &rng)

        for _ in 0..<roots where !free.isEmpty {
            tiles[free.removeLast()] = .root
        }
        // Rocks may land on the find, so put it back in the pool first.
        var rockPool = free + [findTile]
        rockPool.shuffle(using: &rng)
        for _ in 0..<rocks where !rockPool.isEmpty {
            tiles[rockPool.removeLast()] = .rock(hp: 2)
        }
        return tiles
    }

    private static func applyGlint(_ tiles: inout [DigTile], findTile: Int,
                                   condition: DigCondition?,
                                   rng: inout some RandomNumberGenerator) {
        guard condition == .glinting else { return }
        func eligible(_ d: Int) -> [Int] {
            tiles.indices.filter {
                $0 != findTile && tiles[$0].isDiggable && DigGrid.distance($0, findTile) == d
            }
        }
        let pool = eligible(2).isEmpty ? eligible(1) : eligible(2)
        guard let pick = pool.randomElement(using: &rng) else { return }
        tiles[pick] = .broken        // a rock here breaks fully, per the spec
    }

    /// At least 4 diggable tiles within d ≤ 2 of the find, and on a wet spot at least 2 of
    /// them at d == 1 (because d == 2 reads as cold there).
    private static func hasEnoughClues(_ tiles: [DigTile], findTile: Int, wet: Bool) -> Bool {
        let near = tiles.indices.filter {
            $0 != findTile && tiles[$0].isDiggable && DigGrid.distance($0, findTile) <= 2
        }
        guard near.count >= 4 else { return false }
        guard wet else { return true }
        return near.filter { DigGrid.distance($0, findTile) == 1 }.count >= 2
    }
}

/// A find waiting in the soil, with the dig grid the player works through to reach it.
struct BuriedFind: Codable, Equatable {
    var spot: Int
    var kind: FindKind
    var condition: DigCondition?
    var findTile: Int
    var tiles: [DigTile]
    /// Set the first time the player digs here; drives the condition badge and the spur.
    var visited: Bool

    /// A grid is "missing" when it came from a v1 save (or a part-written v2 one).
    var needsGrid: Bool { tiles.count != DigGrid.tileCount }

    init(spot: Int, kind: FindKind, condition: DigCondition?,
         findTile: Int, tiles: [DigTile], visited: Bool = false) {
        self.spot = spot; self.kind = kind; self.condition = condition
        self.findTile = findTile; self.tiles = tiles; self.visited = visited
    }

    /// Convenience used by `fresh` and the tests: builds a grid from a seed derived from the
    /// spot, so the same spot always gets the same grid.
    init(spot: Int, kind: FindKind, condition: DigCondition? = nil) {
        var rng = BurrowRNG(seed: BuriedFind.seed(for: spot))
        let g = DigGen.make(layer: BurrowSpots.layer(spot), condition: condition, rng: &rng)
        self.init(spot: spot, kind: kind, condition: condition,
                  findTile: g.findTile, tiles: g.tiles)
    }

    static func seed(for spot: Int) -> UInt64 { 0x5EED_0000 &+ UInt64(spot) }

    /// v1 saves have only `spot` and `kind`. Decoding the rest leniently is what keeps
    /// `BurrowStore.load` from treating every existing save as corrupt.
    init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        spot = try c.decode(Int.self, forKey: .spot)
        kind = try c.decode(FindKind.self, forKey: .kind)
        condition = try c.decodeIfPresent(DigCondition.self, forKey: .condition)
        findTile = try c.decodeIfPresent(Int.self, forKey: .findTile) ?? 0
        tiles = try c.decodeIfPresent([DigTile].self, forKey: .tiles) ?? []
        visited = try c.decodeIfPresent(Bool.self, forKey: .visited) ?? false
    }
}

func load(now: Date = Date()) -> BurrowSave {
    guard let data = try? Data(contentsOf: fileURL) else { return .fresh(now: now) }
    struct VersionOnly: Decodable { let version: Int }
    if let v = try? JSONDecoder().decode(VersionOnly.self, from: data),
       v.version <= BurrowSave.currentVersion,
       var save = try? JSONDecoder().decode(BurrowSave.self, from: data) {
        if migrate(&save) { try? self.save(save) }
        return save
    }
    moveAside()
    return .fresh(now: now)
}

/// Fill in anything an older save is missing. Returns true if the save changed.
/// Grids are rebuilt from a spot-derived seed, so migrating twice gives the same burrow.
private func migrate(_ s: inout BurrowSave) -> Bool {
    var changed = s.version != BurrowSave.currentVersion
    for i in s.buried.indices where s.buried[i].needsGrid {
        let f = s.buried[i]
        var rng = BurrowRNG(seed: BuriedFind.seed(for: f.spot))
        let condition = DigGen.rollCondition(rng: &rng)
        let g = DigGen.make(layer: BurrowSpots.layer(f.spot), condition: condition, rng: &rng)
        // `kind` is kept exactly as it was, even where it falls outside the layer's table.
        s.buried[i] = BuriedFind(spot: f.spot, kind: f.kind, condition: condition,
                                 findTile: g.findTile, tiles: g.tiles, visited: f.visited)
        changed = true
    }
    s.version = BurrowSave.currentVersion
    return changed
}

    /// Strikes a trip to `spot` would grant right now.
    static func tripStrikes(_ s: BurrowSave, spot: Int) -> Int {
        let layer = BurrowSpots.layer(spot)
        let condition = s.buried.first { $0.spot == spot }?.condition
        let fromCondition: Int

        case .wet:   fromCondition = 2
        case .rooty: fromCondition = -1
        default:     fromCondition = 0
        }
        // `isHappy` / `isSulking` are both larva-only, so an egg gets 0. Sulking wins when
        // both hold — a starving grub digs badly however cheerful it is.
        let fromMood = isSulking(s) ? -1 : isHappy(s) ? 2 : 0
        return max(2, layer.strikes + fromCondition + fromMood)
    }

    /// Spend a dig charge and walk to the spot. Mood is read once, here.
    static func startTrip(_ s0: BurrowSave, spot: Int) -> BurrowResult {
        guard s0.stage == .egg || s0.stage == .larva else { return refuse(s0, .wrongStage) }
        guard let i = s0.buried.firstIndex(where: { $0.spot == spot })
        else { return refuse(s0, .nothingBuriedThere) }
        guard s0.digs > 0 else { return refuse(s0, .noDigs) }

        var s = s0
        s.digs -= 1
        s.buried[i].visited = true
        return BurrowResult(save: s, events: [.tripStarted(spot: spot, strikes: tripStrikes(s, spot: spot))])
    }
}
```

Note `tripStrikes` is called **after** the charge is spent but reads only stats, so the order doesn't matter; keeping it public lets the view show the count before the trip starts.

```swift
/// One shovel strike at `tile`. `strikesLeft` is passed in only so the rules can refuse;
/// the caller decrements it whenever the result is not a refusal.
static func strike(_ s0: BurrowSave, spot: Int, tile: Int, strikesLeft: Int) -> BurrowResult {
    guard (0..<DigGrid.tileCount).contains(tile) else { return refuse(s0, .badTile) }
    guard let i = s0.buried.firstIndex(where: { $0.spot == spot })
    else { return refuse(s0, .nothingBuriedThere) }
    guard strikesLeft > 0 else { return refuse(s0, .noStrikes) }

    let find = s0.buried[i]
    switch find.tiles[tile] {
    case .root:   return refuse(s0, .rootTile)
    case .broken: return refuse(s0, .alreadyBroken)
    case .rock(let hp) where hp > 1:
        var s = s0
        s.buried[i].tiles[tile] = .rock(hp: hp - 1)
        return BurrowResult(save: s, events: [.rockChipped(tile: tile)])
    case .soil, .rock:
        var s = s0
        s.buried[i].tiles[tile] = .broken
        let d = DigGrid.distance(tile, find.findTile)
        var events: [BurrowEvent] = [
            .tileBroken(tile: tile, heat: DigGrid.heat(distance: d, wet: find.condition == .wet))
        ]
        if tile == find.findTile {
            s.buried.remove(at: i)
            s.holes.append(spot)
            s.bag[find.kind, default: 0] += 1
            events.append(.dugUp(spot: spot, kind: find.kind))
        }
        return BurrowResult(save: s, events: events)
    }
}
```

```swift
import Foundation

/// A deterministic greedy player, used by the pacing test and the balance sim — never by the
/// game. Spec §6 pins its behaviour, because the pacing figures depend on it.
enum DigSolver {
    /// Would a player who always makes the locally best move find the item in `strikes`?
    static func solves(tiles: [DigTile], findTile: Int, wet: Bool, strikes: Int) -> Bool {
        var tiles = tiles
        var left = strikes
        // Every tile that could still hold the find, given what has been revealed.
        var candidates = Set(tiles.indices.filter { tiles[$0] != .root })

        while left > 0 {
            guard let pick = choose(tiles: tiles, candidates: candidates) else { return false }
            let cost = tiles[pick].cost
            if cost > left { return false }
            left -= cost
            tiles[pick] = .broken
            if pick == findTile { return true }
            candidates.remove(pick)
            // Keep only the candidates consistent with the heat just revealed.
            let heat = DigGrid.heat(distance: DigGrid.distance(pick, findTile), wet: wet)
            candidates = candidates.filter {
                DigGrid.heat(distance: DigGrid.distance(pick, $0), wet: wet) == heat
            }
        }
        return false
    }

    /// Highest (expected candidates eliminated ÷ strike cost); ties go to the lowest index.
    private static func choose(tiles: [DigTile], candidates: Set<Int>) -> Int? {
        var best: (score: Double, tile: Int)?
        for tile in tiles.indices where tiles[tile].isDiggable {
            let score = expectedElimination(tile: tile, candidates: candidates)
                      / Double(tiles[tile].cost)
            if best == nil || score > best!.score + 1e-9 {
                best = (score, tile)
            }
        }
        return best?.tile
    }

    /// Averaged over the candidates, each equally likely to hold the find.
    private static func expectedElimination(tile: Int, candidates: Set<Int>) -> Double {
        guard !candidates.isEmpty else { return 0 }
        var buckets: [Heat: Int] = [:]
        for c in candidates {
            buckets[DigGrid.heat(distance: DigGrid.distance(tile, c), wet: false), default: 0] += 1
        }
        let n = Double(candidates.count)
        // Expected size of the surviving bucket, weighted by the chance of landing in it.
        let expectedRemaining = buckets.values.reduce(0.0) { $0 + Double($1) * Double($1) } / n
        return n - expectedRemaining
    }
}
```

**Note the simplification:** `expectedElimination` ignores wetness when bucketing, so the solver is slightly pessimistic on wet spots. That's deliberate — it keeps the choice function independent of the condition and the pacing figures conservative. Say so in a comment.
